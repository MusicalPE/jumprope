rules_version = '2';
// 점퍼스 클라우드판 Firestore 보안 규칙
// 파이어베이스 콘솔 → Firestore Database → 규칙 에 이 내용을 모두 붙여 넣고 [게시]
// - 선생님: 구글 로그인 + 그 공간의 members 에 있어야 함
// - 학생 기기: 익명 로그인. 공간 주소(24자 무작위)를 아는 기기만 읽을 수 있고, 학생 이름은 암호문으로만 저장됨
// - 학생 기록은 inbox 에 '만들기'만 가능하고, 쓰기표(QR 안에 있음)가 맞아야 함
service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() { return request.auth != null; }
    function google() { return signedIn() && request.auth.token.firebase.sign_in_provider != 'anonymous'; }
    function isMember(s) { return google() && exists(/databases/$(database)/documents/spaces/$(s)/members/$(request.auth.uid)); }
    function isOwner(s) { return google() && get(/databases/$(database)/documents/spaces/$(s)).data.owner == request.auth.uid; }

    function validEntry(s) {
      let d = request.resource.data;
      return d.keys().hasOnly(['s', 'd', 'k', 'n', 'src', 'st', 'w', 't'])
        && d.s is string && d.s.size() <= 16
        && d.d is string && d.d.matches('^20[0-9]{2}-[01][0-9]-[0-3][0-9]$')
        && d.k is string && d.k.size() <= 8
        && d.n is int && d.n >= 1 && d.n <= 5000
        && d.src in ['hand', 'cam', 'rope', 'live']
        && d.st in ['ok', 'wait']
        && d.t == request.time
        && d.w is string && d.w.size() <= 40
        && exists(/databases/$(database)/documents/spaces/$(s)/keys/$(d.w))
        && (d.st == 'wait' || d.src != 'hand' || isMember(s)
            || get(/databases/$(database)/documents/spaces/$(s)/meta/pub).data.approval != true);
    }

    // 내 공간 목록 (나만)
    match /users/{uid} {
      allow read, write: if google() && request.auth.uid == uid;
    }

    // 선생님 초대 코드: 코드를 아는 사람만 하나씩 열어 볼 수 있음 (목록 보기 불가)
    match /invites/{code} {
      allow get: if google();
      allow create: if isMember(request.resource.data.sid) && request.resource.data.keys().hasOnly(['sid', 'by', 'exp']);
      allow delete: if google() && isMember(resource.data.sid);
    }

    match /spaces/{s} {
      allow create: if google() && request.resource.data.owner == request.auth.uid;
      allow get: if isMember(s);
      // 선생님은 이름·쓰기표만 바꿀 수 있고, 주인·잠금 열쇠는 주인만
      allow update: if isMember(s)
        && request.resource.data.owner == resource.data.owner
        && (isOwner(s) || !request.resource.data.diff(resource.data).affectedKeys().hasAny(['owner', 'crypto']));
      allow delete: if isOwner(s);

      match /members/{uid} {
        allow read: if isMember(s);
        allow create: if google() && request.auth.uid == uid && (
          (get(/databases/$(database)/documents/spaces/$(s)).data.owner == uid && request.resource.data.role == 'owner')
          || (request.resource.data.role == 'teacher'
              && request.resource.data.code is string
              && get(/databases/$(database)/documents/invites/$(request.resource.data.code)).data.sid == s
              && get(/databases/$(database)/documents/invites/$(request.resource.data.code)).data.exp > request.time.toMillis()));
        allow update: if isOwner(s);
        allow delete: if google() && (request.auth.uid == uid || isOwner(s));
      }

      // 공개 설정(학교 이름·승인 절차·종류)과 암호문 명단: 공간 주소를 아는 기기만
      match /meta/{m} {
        allow get: if signedIn();
        allow write: if isMember(s);
      }
      // 달 요약
      match /months/{ym} {
        allow get: if signedIn();
        allow write: if isMember(s);
      }
      // 쓰기표: 규칙 안에서 exists() 로만 확인 (아무도 읽을 수 없음)
      match /keys/{w} {
        allow read: if false;
        allow write: if isMember(s);
      }
      // 학생이 넣은 기록
      match /inbox/{id} {
        allow get, list: if signedIn();
        allow create: if signedIn() && validEntry(s);
        allow update, delete: if isMember(s);
      }
    }
  }
}
