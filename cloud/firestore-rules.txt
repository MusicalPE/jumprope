rules_version = '2';
// 점퍼스 클라우드판 Firestore 보안 규칙
// 파이어베이스 콘솔 → Firestore Database → 규칙 에 이 내용을 모두 붙여 넣고 [게시]
// - 선생님: 구글 로그인 + 그 공간의 members 에 있어야 함
// - 학생 기기: 익명 로그인. 공간 주소(24자 무작위)를 아는 기기만 읽을 수 있고, 학생 이름은 암호문으로만 저장됨
// - 학생 기록은 inbox 에 '만들기'만 가능하고, 쓰기표(QR 안에 있음)가 맞아야 함. 기록 줄도 암호문(box)이고 횟수(n)만 규칙이 확인함
service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() { return request.auth != null; }
    function google() { return signedIn() && request.auth.token.firebase.sign_in_provider != 'anonymous'; }
    function isMember(s) { return google() && exists(/databases/$(database)/documents/spaces/$(s)/members/$(request.auth.uid)); }
    function isOwner(s) { return google() && get(/databases/$(database)/documents/spaces/$(s)).data.owner == request.auth.uid; }
    // 운영자: 파이어베이스 콘솔에서 admins/{계정 UID} 문서를 만들어 둔 계정 (앱에서는 만들 수 없음)
    function isAdmin() { return google() && exists(/databases/$(database)/documents/admins/$(request.auth.uid)); }
    // 새 공간 만들기: 학교 이름이 운영자가 올린 학교 목록(gate/schools)에 있고 · 중지되지 않았고 · 가입 암호(gate/main 의 hash)가 맞아야 함
    function gateOk(d) {
      return d.gate is string
        && d.gate == get(/databases/$(database)/documents/gate/main).data.hash
        && d.name in get(/databases/$(database)/documents/gate/schools).data.names
        && !(d.name in get(/databases/$(database)/documents/gate/schools).data.get('off', []));
    }

    function validEntry(s) {
      let d = request.resource.data;
      return d.keys().hasOnly(['box', 'n', 'src', 'st', 'w', 't'])
        && d.box is map && d.box.ct is string && d.box.ct.size() <= 6000
        && d.n is int && d.n >= 1 && d.n <= 5000
        && d.src in ['hand', 'cam', 'rope', 'live']
        && d.st in ['ok', 'wait']
        && d.t == request.time
        && d.w is string && d.w.size() <= 40
        && exists(/databases/$(database)/documents/spaces/$(s)/keys/$(d.w))
        && (d.st == 'wait' || d.src != 'hand' || isMember(s)
            || get(/databases/$(database)/documents/spaces/$(s)/meta/pub).data.approval != true);
    }

    // 내 공간 목록 (나만)
    match /users/{uid} {
      allow read, write: if google() && request.auth.uid == uid;
    }

    // 운영자 표시: 자기 것만 읽음 (콘솔에서만 만들고 지움)
    match /admins/{uid} {
      allow get: if google() && request.auth.uid == uid;
      allow list, write: if false;
    }
    // 가입 문: 학교 목록(schools)은 로그인한 선생님이 읽고, 가입 암호(main)는 운영자만 읽음. 고치는 것은 운영자만
    match /gate/{g} {
      allow get: if isAdmin() || (google() && g == 'schools');
      allow list: if false;
      allow write: if isAdmin();
    }

    // 선생님 초대 코드: 코드를 아는 사람만 하나씩 열어 볼 수 있음 (목록 보기 불가)
    match /invites/{code} {
      allow get: if google();
      allow create: if isMember(request.resource.data.sid) && request.resource.data.keys().hasOnly(['sid', 'by', 'exp']);
      allow delete: if google() && isMember(resource.data.sid);
    }

    match /spaces/{s} {
      // 새 공간은 학교 목록에 있는 이름 + 가입 암호가 맞아야 만들 수 있음 (운영자는 그냥 만듦)
      allow create: if google() && request.resource.data.owner == request.auth.uid
        && (isAdmin() || gateOk(request.resource.data));
      allow get: if isMember(s);
      allow list: if isAdmin();   // 운영자 화면의 학교별 공간 수
      // 선생님은 쓰기표만 바꿀 수 있고, 주인·잠금 열쇠는 주인만. 학교 이름은 운영자만
      // 주인 넘기기(3.2.0): 지금 주인만, 이 공간의 선생님(members)에게만
      allow update: if isMember(s)
        && (request.resource.data.owner == resource.data.owner
            || (isOwner(s) && exists(/databases/$(database)/documents/spaces/$(s)/members/$(request.resource.data.owner))))
        && (isOwner(s) || !request.resource.data.diff(resource.data).affectedKeys().hasAny(['owner', 'crypto']))
        && (isAdmin() || !request.resource.data.diff(resource.data).affectedKeys().hasAny(['name', 'code', 'gate']));
      allow delete: if isOwner(s);

      match /members/{uid} {
        allow read: if isMember(s);
        allow create: if google() && request.auth.uid == uid && (
          (get(/databases/$(database)/documents/spaces/$(s)).data.owner == uid && request.resource.data.role == 'owner')
          || (request.resource.data.role == 'teacher'
              && request.resource.data.code is string
              && get(/databases/$(database)/documents/invites/$(request.resource.data.code)).data.sid == s
              && get(/databases/$(database)/documents/invites/$(request.resource.data.code)).data.exp > request.time.toMillis()));
        allow update: if isOwner(s);
        allow delete: if google() && (request.auth.uid == uid || isOwner(s));
      }

      // 공개 설정(학교 이름·승인 절차): 공간 주소를 아는 기기만
      // meta/flags 는 운영자가 파이어베이스 콘솔에서만 만들고 고침 (live: true → 전용 점프 라이브 사용). 앱에서는 아무도 못 씀
      match /meta/{m} {
        allow get: if signedIn();
        // 공개 설정의 학교 이름은 공간 문서의 이름과 같아야 함 (학교 목록에서 고른 이름 그대로)
        allow create, update: if isMember(s) && m != 'flags'
          && (m != 'pub' || isAdmin() || request.resource.data.name == get(/databases/$(database)/documents/spaces/$(s)).data.name);
        allow delete: if isMember(s) && m != 'flags';
      }
      // 표(학생·기록·급수·설정) — 모두 암호문. 공간 주소를 아는 기기만 읽고, 선생님만 씀
      match /book/{d} {
        allow get, list: if signedIn();
        allow write: if isMember(s);
      }
      // 선생님만 읽는 설정(전국 현황판 학교 키·AI 열쇠) — 암호문
      match /priv/{d} {
        allow read, write: if isMember(s);
      }
      // 쓰기표: 규칙 안에서 exists() 로만 확인 (아무도 읽을 수 없음)
      match /keys/{w} {
        allow read: if false;
        allow write: if isMember(s);
      }
      // 학생이 넣은 기록
      match /inbox/{id} {
        allow get, list: if signedIn();
        allow create: if signedIn() && validEntry(s);
        allow update, delete: if isMember(s);
      }
    }
  }
}
